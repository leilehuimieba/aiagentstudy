class Knap
{

       Typep Bound(int i);  //Ͻ

       Typew c;  //
       int n;  //Ʒ

       Typew cw;  //ǰ
       Typep cp;  //ǰֵ
       Typep bestp;  //ǰżֵ
};

void Knap<Typew,Typep>::Backtrack(int i)
{   if(i>n)    {     bestp=cp;     return;    }
    if(cw+w[i]<=c)  //
    {  cw+=w[i];
        cp+=p[i];
        Backtrack(i+1);
        cw-=w[i];
        cp-=p[i];    }
    if(Bound(i+1)>bestp) //
        Backtrack(i+1);
}

Typep Knap<Typew,Typep>::Bound(int i)
{
    Typew cleft=c-cw;  //ʣı
    Typep b=cp;  //bΪǰֵ
    //װ뵥λֵߵƷ
    while(i<=n&&w[i]<=cleft)
    {  cleft-=w[i];        b+=p[i];        i++;    }
    if(i<=n)  //װƷһ
        b+=p[i]*cleft/w[i];
    return b;  //Ͻ
}
